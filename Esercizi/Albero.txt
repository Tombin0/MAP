Tipi: Albero, Nodo, Intero, Boolean

Operatori:

-aggiungiS(Albero, Nodo, Nodo)Albero // aggiunge il secondo nodo come figlio sinistro del primo nodo nell’albero

-aggiungiD(Albero, Nodo, Nodo)Albero // aggiunge il secondo nodo come figlio destro del primo nodo nell’albero

	conta(crea)=0
	conta(aggiungiR(a,n1))=1
	conta(aggiungiS(a,n1))=1+conta(A)
	conta(aggiungiD(a,n1))=1+conta(A)
